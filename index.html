const MENUS = {}; async function loadMenus() { const response = await fetch("./menus/manifest.json"); if (!response.ok) { throw new Error("Could not load menu manifest"); } const files = await response.json(); const weeks = await Promise.all( files.map(async (filename) => { const module = await import(`./menus/${filename}`); return module.default; }) ); for (const week of weeks) { for (const [date, menu] of Object.entries(week)) { if (Object.hasOwn(MENUS, date)) { throw new Error(`Duplicate menu date: ${date}`); } MENUS[date] = menu; } } }
